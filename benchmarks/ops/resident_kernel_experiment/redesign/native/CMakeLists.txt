cmake_minimum_required(VERSION 3.16)
project(resident_redesign LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
set(RUN_MODE npu CACHE STRING "AscendC run mode")
if(NOT SOC_VERSION OR NOT ASCEND_HOME_PATH OR NOT TORCH_NPU_PATH)
    message(FATAL_ERROR "Pass SOC_VERSION, ASCEND_HOME_PATH and TORCH_NPU_PATH via build.py")
endif()
string(TOLOWER "${SOC_VERSION}" lower_soc)
if(NOT lower_soc MATCHES "ascend910b")
    message(FATAL_ERROR "This experiment currently targets the Ascend910B family only")
endif()
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
foreach(suffix tools/tikcpp/ascendc_kernel_cmake compiler/tikcpp/ascendc_kernel_cmake ascendc_devkit/tikcpp/samples/cmake)
    if(EXISTS "${ASCEND_HOME_PATH}/${suffix}/ascendc.cmake")
        set(ASCENDC_CMAKE_DIR "${ASCEND_HOME_PATH}/${suffix}")
        break()
    endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "Cannot locate CANN AscendC CMake helper")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
# Exactly one explicitly named device entry per translation unit.
ascendc_library(resident_redesign_kernels SHARED build_table.cpp lookup.cpp resolve_copy.cpp batched_copy.cpp)
find_package(Torch REQUIRED)
add_library(resident_redesign_ops SHARED binding.cpp)
target_include_directories(resident_redesign_ops PRIVATE ${TORCH_INCLUDE_DIRS} "${TORCH_NPU_PATH}/include" "${ASCEND_HOME_PATH}/include")
separate_arguments(torch_flags NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(resident_redesign_ops PRIVATE ${torch_flags})
target_link_directories(resident_redesign_ops PRIVATE "${TORCH_NPU_PATH}/lib" "${ASCEND_HOME_PATH}/lib64")
target_link_libraries(resident_redesign_ops PRIVATE ${TORCH_LIBRARIES} torch_npu ascendcl resident_redesign_kernels)
set_target_properties(resident_redesign_ops PROPERTIES BUILD_RPATH "$ORIGIN;$ORIGIN/lib;${TORCH_NPU_PATH}/lib;${ASCEND_HOME_PATH}/lib64")
target_link_options(resident_redesign_ops PRIVATE "-Wl,-rpath,$ORIGIN:$ORIGIN/lib")
